```go
func minWindow(s string, t string) string {
    nS, nT := len(s), len(t)
    if nS < nT { return "" }

    var arrT, arrS [128]int
    for i := 0; i < nT; i++ {
        arrT[t[i]]++
    }

    // 统计 t 中共有多少种不同的字符需要达标
    needKind := 0
    for _, count := range arrT {
        if count > 0 { needKind++ }
    }

    l, r := 0, 0
    matchKind := 0 // 当前窗口内已经达标（数量够了）的字符种类数
    minLen := nS + 1
    resLeft, resRight := 0, 0

    for r < nS {
        // --- 1. 加入当前字符，更新窗口状态 ---
        curR := s[r]
        arrS[curR]++
        // 如果这个字符的数量刚好达到了 t 中的要求，达标种类 +1
        if arrS[curR] == arrT[curR] {
            matchKind++
        }

        // --- 2. 判断是否达标，并尝试缩小左边界（这就是“判断重复/冲突”的变体） ---
        // 只要当前窗口已经覆盖了所有字符，我们就尝试收缩来寻找“最小”
        for matchKind == needKind {
            // 结算结果：如果当前窗口更小，记录下来
            if r-l+1 < minLen {
                minLen = r-l+1
                resLeft, resRight = l, r
            }

            // 移动左边界，尝试缩窗
            curL := s[l]
            // 如果踢出的字符是必须的，且踢出后不再达标，则达标种类 -1
            if arrS[curL] == arrT[curL] {
                matchKind--
            }
            arrS[curL]--
            l++
        }

        // --- 3. 右指针步进 ---
        r++
    }

    if minLen > nS {
        return ""
    }
    return s[resLeft : resRight+1]
}
```

```go
func maxSlidingWindow(nums []int, k int) []int {
    n := len(nums)
    res := make([]int,0, n-k+1)
    l := 0
    r := 0
    // 队列记录的是索引
    que := make([]int, 0, n-k+1)
    for l <= r && r < n {
        // 先判断是否重复(也就是长度过大)
        if r-l+1 > k {
            // 判断是否需要出队
            if que[0] == l {
                que = que[1:]
            }
            l++
        }

        // 维护单调递减序列,对头元素是最大的元素,所以要从队尾处理
        for len(que) > 0 && nums[que[len(que)-1]] <= nums[r]{
            que = que[:len(que)-1]
        }
        que = append(que, r)
        if r-l+1 == k {
            res = append(res, nums[que[0]])
        }
        r++
    }
    return res
}
```

```go
func diameterOfBinaryTree(root *TreeNode) int {
    res := 0
	
    var helper func(root *TreeNode) int
    helper = func(root *TreeNode) int {
        if root == nil {
            return 0
        }

        left := helper(root.Left)
        right := helper(root.Right)

        // 直径：左高度 + 右高度（边数）
        res = max(res, left+right)

        // 返回高度

    }

    return res
```

```go
    if len(nums1) > len(nums2) {

    if len(nums1) == 0 {
```

```go
func (this *MinStack) Push(val int) {
	if len(this.stack) == 0 {
		this.stack = append(this.stack, val)
		this.minStack = append(this.minStack, val)
```

```go
	mystack := make([]int, 0, n)
	res := make([]int, n)
	for i := 0; i < n; i++ {
        // 如果当前温度比栈顶温度高,那就弹出栈

		}

	}
```

31.下一个全排列

思路:

从右向左寻找第一个“下降点'   把下降点的左一元素计为 left

再次右向左寻找第一个把 left 大的元素计为 right

然后交换left和right,然后再反转 left之后的所有元素reverse(nums[left+1:])

```go
func nextPermutation(nums []int) {
    n := len(nums)
    if n <= 1 {
        return
    }

    // 1. 从最后一个元素 (n-1) 开始向前扫描
    i := n - 1
    // 从右向左寻找第一个“下降点
    for i > 0 && nums[i-1] >= nums[i] {
        i--
    }

    // i-1 就是我们要找的“转折点”（即 pivot）
    if i > 0 {
        pivot := i - 1
        // 2. 从后往前找第一个比 nums[pivot] 大的数（接班人）
        j := n - 1
        for j >= i && nums[j] <= nums[pivot] {
            j--
        }
        // 交换转折点和接班人
        nums[pivot], nums[j] = nums[j], nums[pivot]
    }

    // 3. 将转折点之后的部分（即从索引 i 开始到最后）进行反转
    // 因为这部分现在依然是降序，反转后变为升序，能保证增量最小
    reverse(nums[i:])
}

func reverse(a []int) {
    for l, r := 0, len(a)-1; l < r; l, r = l+1, r-1 {
        a[l], a[r] = a[r], a[l]
    }
}
```

```go
            i++
```

```go
func sortList(head *ListNode) *ListNode {
	var mergeSort func(items *ListNode) *ListNode
	mergeSort = func(items *ListNode) *ListNode {
		if items == nil || items.Next == nil {
			return items
		}
		// 找中点
		slow := items
		fast := items
		for fast.Next != nil && fast.Next.Next != nil {
			slow = slow.Next
			fast = fast.Next.Next
		}
		// 中点是 slow
		left := items
		right := slow.Next
		// 断开
		slow.Next = nil
		// 进行排序左边
		left = mergeSort(left)
		// 同理
		right = mergeSort(right)

		return merge(left, right)
	}
	return mergeSort(head)
}
func merge(left, right *ListNode) *ListNode {
	dummy := &ListNode{}
	cur := dummy
	for left != nil && right != nil {
		if left.Val < right.Val {
			// 不需要创建新节点,因为这是排序已有的结点
			cur.Next = left
			left = left.Next
			cur = cur.Next
		} else {
			cur.Next = right
			right = right.Next
			cur = cur.Next
		}
	}
	if left != nil {
		cur.Next = left
	}
	if right != nil {
		cur.Next = right
	}
	return dummy.Next
}
```

```go
func sortArray(nums []int) []int {
    var quickSort func(start, end int)

        if start >= end {
            return
        }
        pivotal := partition(nums, start, end)
        quickSort(start, pivotal-1)
        quickSort(pivotal+1, end)
    }
    quickSort(0, len(nums)-1)
    return nums
}
func partition(nums []int, start, end int) int {
    // 1. 随机化，防止有序数组 TLE
    randIdx := rand.Intn(end-start+1) + start
    nums[start], nums[randIdx] = nums[randIdx], nums[start]

    pivotVal := nums[start]
    l, r := start, end
    // l == r 的时候退出
    for l < r {
        // 必须先动 r
        for l < r && nums[r] >= pivotVal {
            r--
        }
        // 再动 l
        for l < r && nums[l] <= pivotVal {
            l++
        }
        if l < r {
            nums[l], nums[r] = nums[r], nums[l]
        }
    }
    // 基准归位,此时 l == r
    nums[start], nums[r] = nums[r], nums[start]
    return r
}
```

```go
    mapNumsCount := make(map[int]int, n/2)
    for _, num := range nums{
        mapNumsCount[num]++

    k = n-k
    uniNums := make([]int, 0, n)

        uniNums = append(uniNums, num)

            return

    return uniNums[k:]
```

## Constructor

LRU 需要根据key O(1)获取,所以需要 一个 map[int]*Node, Node是为了O(1)来删除

然后为了方便操作,还需要一个虚拟头和虚拟尾

方法思想:

1. get:如果存在先找到node,然后删除,然后再插入头
2. Remove: 如果传来的key 是-1,那就是从尾巴删除一个,否则就 删除对应的key,记得 map里面也要删除. 坑点是 如果只有虚拟头和虚拟尾
3. Put:  如果存在,需要先 Remove,然后再插入头;  如果不存在,直接插入头; 插入头之后,查看是否超过了容量.
4. 我不想优化了:如果可以优化的话,那就是 Put,只修改这个结点的值,然后把这个结点移动到头结点(先删除,再插入到头结点);  但是我之前的方法是直接创建新的,而不是利用原来的结点

```go

type LRUCache struct {
	Capacity    int
	MapLRU      map[int]*Node
	Dummy, Tail *Node // 虚拟头和虚拟尾
}
type Node struct {
	Key, Value int
	Pre, Next  *Node
}

func Constructor(capacity int) LRUCache {
	// +1 防止底层扩容
	maplru := make(map[int]*Node, capacity+1)
	dummy, tail := &Node{}, &Node{}
	dummy.Next = tail
	tail.Pre = dummy

	lrucache := LRUCache{}
	lrucache.Capacity = capacity
	lrucache.MapLRU = maplru
	lrucache.Dummy = dummy
	lrucache.Tail = tail
	return lrucache
}

func (this *LRUCache) Get(key int) int {
	if node, ok := this.MapLRU[key]; ok {
		key := node.Key
		value := node.Value
		// 删除
		this.Remove(key)
		// 放入队首
		this.Put(key, value)
		return value
	}

	// 不存在这个 key
	return -1
}

// 正常 key  是 >= 0的, key < 0 说明需要从队尾删除
func (this *LRUCache) Remove(key int) {
    // 如果链表已经不能删除了,只有虚拟头和虚拟尾了
    if this.Tail.Pre == this.Dummy {
        return
    }
	// 从队尾删除
	if key < 0 {
        tail := this.Tail
        pre := tail.Pre.Pre
        deleteNode := tail.Pre
		pre.Next = tail
		tail.Pre = pre
		delete(this.MapLRU, deleteNode.Key)
		return
	}
	// 正常删除,在队伍中间,也就是根据 map 获取位置,然后删除
	node := this.MapLRU[key]
	pre := node.Pre
	next := node.Next
	pre.Next = next
	next.Pre = pre
	// 不要忘记删除 MapLRU[key]
	delete(this.MapLRU, key)
	return
}
func (this *LRUCache) Put(key int, value int) {
	// 关键修正：如果 key 已经存在，先移除旧节点（包括 Map 和 链表中的位置）
	if _, ok := this.MapLRU[key]; ok {
		this.Remove(key)
	}
	dummy := this.Dummy
	newNode := &Node{Key: key, Value: value}
	next := dummy.Next

	dummy.Next = newNode
	newNode.Pre = dummy
	newNode.Next = next
	next.Pre = newNode
	// 不要忘记加入 MapLRU
	this.MapLRU[key] = newNode
	if this.Capacity < len(this.MapLRU) {
		this.Remove(-1)
	}
}
```

```go
func canFinish(numCourses int, prerequisites [][]int) bool {
	inDegree := make([]int, numCourses)
    // 和 prerequisites 刚好相反,相当于把索引当作前驱用
	graph := make([][]int, numCourses)
    
    // 是把 prerequisites 弄反,所以长度是 len(prerequisites)
    for i := 0; i < len(prerequisites); i++ {
        needPre := prerequisites[i][0]
        cur := prerequisites[i][1]
        // 入度为0的索引不需要++,有前驱的需要++
        inDegree[cur]++
        graph[needPre] = append(graph[needPre], cur)
    }

    myqueue := make([]int, 0)
    for i := 0; i < numCourses; i++ {
        if inDegree[i] == 0 {
            myqueue = append(myqueue, i)
        }
    }

    for len(myqueue) > 0 {
        // 出队
        pre := myqueue[0]
        myqueue = myqueue[1:]

        // 减去当前结点是前驱的结点
        for _, cur := range graph[pre] {
            inDegree[cur]--
            // 此时判断当前减去前驱的结点的入度是否为0
            if inDegree[cur] == 0 {
                myqueue = append(myqueue, cur)
            }
        } 
    }

    for i := 0; i < numCourses; i++ {
        if inDegree[i] != 0 {
            return false
        }
    }
    return true

}
```

```go
func orangesRotting(grid [][]int) int {
	m := len(grid)
	n := len(grid[0])
	// 统计新鲜橘子个数
	countFresh := 0
	// 队列
	myqueue := make([][2]int, 0)
	for i := 0; i < m; i++ {
		for j := 0; j < n; j++ {
			if grid[i][j] == 1 {
				countFresh++
			}
			if grid[i][j] == 2 {
				myqueue = append(myqueue, [2]int{i, j})
			}
		}
	}

	if countFresh == 0 {
		return 0
	}
	// 四个方向
	dirR := [4]int{1, -1, 0, 0}
	dirC := [4]int{0, 0, 1, -1}
    minutes := 0
	for len(myqueue) > 0 {
		length := len(myqueue)
		// 判断当前坏句子是否扩散了
        flag := false
		for i := 0; i < length; i++ {
            // 出队
			r := myqueue[0][0]
			c := myqueue[0][1]
            myqueue = myqueue[1:]
            for j := 0; j < 4; j++ {
                newR := r+dirR[j]
                newC := c+dirC[j]
                if newR >= m || newR < 0 || newC >= n || newC < 0 {
                    continue
                }
                if grid[newR][newC] == 1 {
                    countFresh--
                    grid[newR][newC] = 2
                    flag = true
                    // 不要忘记了, 坏句子入队
                    myqueue = append(myqueue, [2]int{newR, newC})
                }
            }
		}
        // 如果扩散了
        if flag {
            minutes++
        }
	}
    // 扩散完如果还有新鲜橘子
    if countFresh > 0 {
        return -1
    }
    return minutes
}
```
